from http.server import BaseHTTPRequestHandler import json import smtplib from email.mime.multipart import MIMEMultipart from email.mime.text import MIMEText import os # Outlook SMTP Details SMTP_SERVER = "smtp.office365.com" SMTP_PORT = 587 SENDER_EMAIL = os.environ.get("OUTLOOK_EMAIL") # e.g., logistics@saharaukfoods.co.uk SENDER_PASSWORD = os.environ.get("OUTLOOK_PASSWORD") # App Password or Account Password class handler(BaseHTTPRequestHandler): def do_POST(self): content_length = int(self.headers['Content-Length']) post_data = self.rfile.read(content_length) data = json.loads(post_data.decode('utf-8')) # Prepare Branded HTML Email Template subject = f"NEW ORDER LOGGED: {data['type']} - {data['client']} ({data['po']})" html_body = f"""

Sahara UK Foods - New Order Notification

A new order has been submitted via the Operations Dashboard:

Logged By{data['user']}
Order Type{data['type']}
Client Name{data['client']}
PO Ref / Order #{data['po']}
Product Name{data['product']}
Quantity{data['qty']}
Country of Origin (COO){data['coo']}
Price{data['price']}

Automated notification sent via Sahara UK Foods Logistics Operations System.

""" try: msg = MIMEMultipart("alternative") msg["Subject"] = subject msg["From"] = SENDER_EMAIL msg["To"] = "team@saharaukfoods.co.uk" # Receiver team email address msg.attach(MIMEText(html_body, "html")) server = smtplib.SMTP(SMTP_SERVER, SMTP_PORT) server.starttls() server.login(SENDER_EMAIL, SENDER_PASSWORD) server.sendmail(SENDER_EMAIL, msg["To"], msg.as_string()) server.quit() self.send_response(200) self.send_header('Content-type', 'application/json') self.end_headers() self.wfile.write(json.dumps({"status": "success", "message": "Email sent"}).encode('utf-8')) except Exception as e: self.send_response(500) self.send_header('Content-type', 'application/json') self.end_headers() self.wfile.write(json.dumps({"status": "error", "message": str(e)}).encode('utf-8'))
